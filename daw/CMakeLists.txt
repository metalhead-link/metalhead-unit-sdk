# A unit file's plugins as CLAP, VST3 and AUv2 plugins for a DAW on the Mac (see README.md).
#
#   cmake -S . -B build/yafr -G Ninja -DUNIT=yafr && cmake --build build/yafr
#
# UNIT is a unit directory (relative to ccs/units, or absolute); its Makefile says what to
# build, as for the device. The plugins land in build/<unit>/plugins. The Makefile here wraps
# this. clap-wrapper and the CLAP, VST3 and AudioUnit SDKs are downloaded at the first configure.

cmake_minimum_required(VERSION 3.21)

if (NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS version")
endif()
if (NOT DEFINED CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "Universal: DAWs running under Rosetta load them too")
endif()

project(metalhead-unit-plugins VERSION 1.0 LANGUAGES C CXX)

if (NOT APPLE)
    message(FATAL_ERROR "unitclap.c logs with os_log: macOS only for now")
endif()

# --- The unit -------------------------------------------------------------------------------

set(UNIT "" CACHE STRING "Unit directory to build: yafr, drums, stock, ...")
if (UNIT STREQUAL "")
    message(FATAL_ERROR "Set UNIT to a unit directory: -DUNIT=yafr")
endif()
set(UNITS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/..")
get_filename_component(UNIT_DIR "${UNIT}" ABSOLUTE BASE_DIR "${UNITS_DIR}")

execute_process(
    COMMAND make -s --no-print-directory -C "${UNIT_DIR}" -f Makefile -f "${CMAKE_CURRENT_SOURCE_DIR}/unitvars.mk" unitvars
    OUTPUT_VARIABLE UNIT_VARS
    RESULT_VARIABLE UNIT_VARS_RESULT)
if (NOT UNIT_VARS_RESULT EQUAL 0)
    message(FATAL_ERROR "Can't read the unit Makefile in ${UNIT_DIR}")
endif()
foreach (var NAME BUNDLE PLUGINS SRCS)
    string(REGEX MATCH "${var}=([^\n]*)" _ "${UNIT_VARS}")
    separate_arguments(UNIT_${var} UNIX_COMMAND "${CMAKE_MATCH_1}")
endforeach()
if (NOT UNIT_BUNDLE OR NOT UNIT_PLUGINS OR NOT UNIT_SRCS)
    message(FATAL_ERROR "${UNIT_DIR}/Makefile doesn't set BUNDLE, PLUGINS and SRCS")
endif()
list(TRANSFORM UNIT_SRCS PREPEND "${UNIT_DIR}/")
message(STATUS "unit ${UNIT_BUNDLE}: ${UNIT_PLUGINS}")

# unit_entry(), as unit.mk generates it
set(BUNDLE_C "${CMAKE_CURRENT_BINARY_DIR}/bundle.c")
set(bundle "/* Generated by CMakeLists.txt from ${UNIT_DIR}/Makefile */\n#include <unit_abi.h>\n")
foreach (p ${UNIT_PLUGINS})
    string(APPEND bundle "UnitPluginEntry ${p};\n")
endforeach()
list(JOIN UNIT_PLUGINS ", " plugin_list)
string(APPEND bundle "UNIT_BUNDLE(\"${UNIT_BUNDLE}\", ${plugin_list});\n")
file(CONFIGURE OUTPUT "${BUNDLE_C}" CONTENT "${bundle}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${UNIT_DIR}/Makefile")

# --- clap-wrapper ----------------------------------------------------------------------------

set(CLAP_WRAPPER_DOWNLOAD_DEPENDENCIES TRUE CACHE BOOL "Download the CLAP, VST3 and AudioUnit SDKs")
include(FetchContent)
FetchContent_Declare(clap-wrapper
    GIT_REPOSITORY https://github.com/free-audio/clap-wrapper.git
    GIT_TAG v0.17.0
    GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(clap-wrapper)

# --- The plugins ----------------------------------------------------------------------------

# The unit's sources, compiled natively as for `make test`, and the wrapper. Only clap_entry
# is exported, so two unit plugins in one DAW don't see each other's symbols.
add_library(unit-impl STATIC unitclap.c "${BUNDLE_C}" ${UNIT_SRCS})
target_include_directories(unit-impl PUBLIC "${UNITS_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(unit-impl PUBLIC clap clap-wrapper-extensions)
target_compile_options(unit-impl PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall -O2>)
set_target_properties(unit-impl PROPERTIES
    C_STANDARD 11
    C_EXTENSIONS ON
    C_VISIBILITY_PRESET hidden
    CXX_VISIBILITY_PRESET hidden)

make_clapfirst_plugins(
    TARGET_NAME "${UNIT_NAME}"
    IMPL_TARGET unit-impl
    OUTPUT_NAME "metalhead ${UNIT_BUNDLE}"
    ENTRY_SOURCE entry.cpp
    BUNDLE_IDENTIFIER "metalhead.unit.${UNIT_BUNDLE}"
    BUNDLE_VERSION ${PROJECT_VERSION}
    COPY_AFTER_BUILD FALSE
    PLUGIN_FORMATS CLAP VST3 AUV2
    ASSET_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
